import React, { useState } from 'react';
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  StyleSheet,
  SafeAreaView,
  ScrollView,
  Alert
} from 'react-native';

const GOLD = '#D6AD60';
const BG = '#090909';

export default function App() {
  const [screen, setScreen] = useState('login');
  const [tab, setTab] = useState('Keşfet');
  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [confirm, setConfirm] = useState('');
  const [liked, setLiked] = useState([]);

  const people = [
    { id: 1, name: 'Deniz', age: 25, emoji: '🌙', distance: '2 km' },
    { id: 2, name: 'Ege', age: 27, emoji: '🖤', distance: '4 km' },
    { id: 3, name: 'Lina', age: 24, emoji: '✨', distance: '6 km' },
    { id: 4, name: 'Aras', age: 26, emoji: '🌹', distance: '8 km' }
  ];

  const validEmail = () =>
    /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.trim());

  const login = () => {
    if (!validEmail() || password.length < 6) {
      Alert.alert('Bilgileri kontrol et', 'Geçerli e-posta ve en az 6 karakterli şifre gir.');
      return;
    }
    setName(name.trim() || email.split('@')[0]);
    setScreen('home');
    setTab('Keşfet');
  };

  const register = () => {
    if (!name.trim() || !validEmail() || password.length < 6) {
      Alert.alert('Eksik bilgi', 'Adını, geçerli e-postanı ve en az 6 karakterli şifreni gir.');
      return;
    }
    if (password !== confirm) {
      Alert.alert('Şifreler eşleşmiyor', 'Şifrelerini tekrar kontrol et.');
      return;
    }
    setScreen('home');
    setTab('Keşfet');
  };

  const toggleLike = (id) => {
    setLiked(current =>
      current.includes(id)
        ? current.filter(item => item !== id)
        : [...current, id]
    );
  };

  const Button = ({ title, onPress, secondary = false }) => (
    <TouchableOpacity
      style={[styles.button, secondary && styles.secondaryButton]}
      onPress={onPress}
    >
      <Text style={[styles.buttonText, secondary && styles.secondaryText]}>
        {title}
      </Text>
    </TouchableOpacity>
  );

  if (screen === 'login' || screen === 'register') {
    const registering = screen === 'register';

    return (
      <SafeAreaView style={styles.container}>
        <ScrollView contentContainerStyle={styles.auth}>
          <Text style={styles.logo}>YAKIN<Text style={styles.gold}>.</Text></Text>
          <Text style={styles.tagline}>BAĞLANTI KUR. YAKINLAŞ.</Text>

          <View style={styles.panel}>
            <Text style={styles.heading}>
              {registering ? 'Aramıza katıl.' : 'Tekrar hoş geldin.'}
            </Text>
            <Text style={styles.muted}>
              {registering ? 'Yeni hesabını oluştur.' : 'YAKIN dünyasına giriş yap.'}
            </Text>

            {registering && (
              <TextInput
                style={styles.input}
                placeholder="Adın"
                placeholderTextColor="#888"
                value={name}
                onChangeText={setName}
              />
            )}

            <TextInput
              style={styles.input}
              placeholder="E-posta adresin"
              placeholderTextColor="#888"
              value={email}
              onChangeText={setEmail}
              keyboardType="email-address"
              autoCapitalize="none"
            />

            <TextInput
              style={styles.input}
              placeholder="Şifre (en az 6 karakter)"
              placeholderTextColor="#888"
              value={password}
              onChangeText={setPassword}
              secureTextEntry
            />

            {registering && (
              <TextInput
                style={styles.input}
                placeholder="Şifreni tekrar yaz"
                placeholderTextColor="#888"
                value={confirm}
                onChangeText={setConfirm}
                secureTextEntry
              />
            )}

            <Button
              title={registering ? 'KAYIT OL' : 'GİRİŞ YAP'}
              onPress={registering ? register : login}
            />

            <Button
              title={registering ? 'Giriş ekranına dön' : 'Yeni hesap oluştur'}
              onPress={() => setScreen(registering ? 'login' : 'register')}
              secondary
            />
            <Text style={styles.note}>
              Bu sürüm demo giriş/kayıt ekranıdır. Hesaplar henüz sunucuya kaydedilmez.
            </Text>
          </View>
        </ScrollView>
      </SafeAreaView>
    );
  }

  const renderPeople = (list) => (
    list.map(person => (
      <View style={styles.card} key={person.id}>
        <View style={styles.avatar}>
          <Text style={styles.emoji}>{person.emoji}</Text>
        </View>
        <View style={styles.personRow}>
          <View>
            <Text style={styles.personName}>{person.name}, {person.age}</Text>
            <Text style={styles.muted}>Yaklaşık {person.distance}</Text>
          </View>
          <TouchableOpacity onPress={() => toggleLike(person.id)} style={styles.heart}>
            <Text style={styles.heartText}>
              {liked.includes(person.id) ? '♥' : '♡'}
            </Text>
          </TouchableOpacity>
        </View>
      </View>
    ))
  );

  return (
    <SafeAreaView style={styles.container}>
      <View style={styles.header}>
        <Text style={styles.logoSmall}>YAKIN<Text style={styles.gold}>.</Text></Text>
        <Text style={styles.tagline}>BAĞLANTI KUR. YAKINLAŞ.</Text>
      </View>

      <View style={styles.main}>
        {tab === 'Keşfet' && (
          <ScrollView>
            <Text style={styles.heading}>Yakınındakiler</Text>
            <Text style={styles.muted}>Örnek profiller — gerçek kullanıcılar değildir.</Text>
            {renderPeople(people)}
          </ScrollView>
        )}

        {tab === 'Beğeniler' && (
          <ScrollView>
            <Text style={styles.heading}>Beğenilerin</Text>
            {liked.length ? renderPeople(people.filter(p => liked.includes(p.id))) :
              <Text style={styles.empty}>Henüz beğenin yok.</Text>}
          </ScrollView>
        )}

        {tab === 'Mesajlar' && (
          <View>
            <Text style={styles.heading}>Mesajlar</Text>
            <Text style={styles.empty}>Mesajlaşma özelliği sonraki aşamada eklenecek.</Text>
          </View>
        )}

        {tab === 'Profil' && (
          <ScrollView>
            <Text style={styles.heading}>Profilim</Text>
            <Text style={styles.personName}>{name}</Text>
            <Text style={styles.muted}>{email}</Text>
            <Text style={styles.note}>Profil düzenleme ve Premium özellikleri sonraki aşamada eklenecek.</Text>
            <Button title="ÇIKIŞ YAP" onPress={() => {
              setScreen('login');
              setPassword('');
              setConfirm('');
              setTab('Keşfet');
            }} />
          </ScrollView>
        )}
      </View>

      <View style={styles.nav}>
        {['Keşfet', 'Beğeniler', 'Mesajlar', 'Profil'].map(item => (
          <TouchableOpacity key={item} onPress={() => setTab(item)} style={styles.navItem}>
            <Text style={[styles.navText, tab === item && styles.active]}>{item}</Text>
          </TouchableOpacity>
        ))}
      </View>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: BG },
  auth: { flexGrow: 1, justifyContent: 'center', padding: 24 },
  logo: { color: '#fff', fontSize: 42, fontWeight: '900', letterSpacing: 7, textAlign: 'center' },
  logoSmall: { color: '#fff', fontSize: 29, fontWeight: '900', letterSpacing: 5 },
  gold: { color: GOLD },
  tagline: { color: GOLD, fontSize: 10, letterSpacing: 3, textAlign: 'center', marginTop: 8, marginBottom: 35 },
  panel: { backgroundColor: '#151515', borderWidth: 1, borderColor: '#49391F', borderRadius: 20, padding: 20 },
  heading: { color: GOLD, fontSize: 25, fontWeight: 'bold', marginBottom: 10 },
  muted: { color: '#999', fontSize: 13, marginBottom: 14 },
  input: { backgroundColor: '#222', color: '#fff', borderWidth: 1, borderColor: '#49391F', borderRadius: 10, padding: 15, marginTop: 13 },
  button: { backgroundColor: GOLD, borderRadius: 10, padding: 15, alignItems: 'center', marginTop: 18 },
  buttonText: { color: BG, fontWeight: 'bold', letterSpacing: 1 },
  secondaryButton: { backgroundColor: 'transparent', borderWidth: 1, borderColor: GOLD },
  secondaryText: { color: GOLD },
  note: { color: '#888', fontSize: 11, textAlign: 'center', marginTop: 18, lineHeight: 17 },
  header: { padding: 18, borderBottomWidth: 1, borderColor: '#49391F' },
  main: { flex: 1, padding: 18 },
  card: { backgroundColor: '#171717', borderWidth: 1, borderColor: '#49391F', borderRadius: 16, padding: 14, marginTop: 16 },
  avatar: { height: 135, backgroundColor: '#29231D', borderRadius: 12, alignItems: 'center', justifyContent: 'center' },
  emoji: { fontSize: 65 },
  personRow: { flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginTop: 14 },
  personName: { color: '#fff', fontSize: 19, fontWeight: 'bold' },
  heart: { backgroundColor: '#302516', borderRadius: 30, paddingHorizontal: 15, paddingVertical: 8 },
  heartText: { color: GOLD, fontSize: 25 },
  empty: { color: '#bbb', textAlign: 'center', marginTop: 45 },
  nav: { flexDirection: 'row', justifyContent: 'space-around', paddingVertical: 16, backgroundColor: '#151515', borderTopWidth: 1, borderColor: '#49391F' },
  navItem: { paddingHorizontal: 4 },
  navText: { color: '#888', fontSize: 12 },
  active: { color: GOLD, fontWeight: 'bold' }
});
